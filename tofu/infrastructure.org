** Provider
Provider setup: how OpenTofu authenticates to Proxmox.
Authentication tokens should be passed via env var =PROXMOX_VE_API_TOKEN= in the formatted as =user@realm!tokenid=secret-uuid=.
#+begin_src tofu
provider "proxmox" {
  endpoint = "https://andira.lbic.fee.unicamp.br:8006/"
  # Set to true if using self-signed certificates on the Proxmox host
  insecure = true
}

** LXC Container
This block provisions the Debian 13 (Trixie) container on the internal bridge (=vmbr1=).
#+begin_src tofu
resource "proxmox_virtual_environment_container" "headscale" {
  description = "Headscale VPN Controller"
  tags        = ["core", "vpn"]
  node_name = "andira"
  vm_id     = 201

  # ensure unprivileged mode
  unprivileged = true

  # autostart
  start_on_boot = true

  # enable nesting
  features {
    nesting = true
  }

  # ensure the container starts early in the boot sequence (lower = earlier)
  startup {
    order      = 10
    up_delay   = 5
    down_delay = 5
  }
#+end_src
