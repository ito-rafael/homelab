** Providers
*** Proxmox
Provider setup: how OpenTofu authenticates to Proxmox.
Authentication tokens should be passed via env var =PROXMOX_VE_API_TOKEN= in the formatted as =user@realm!tokenid=secret-uuid=.
#+begin_src tofu
provider "proxmox" {
  endpoint = "https://andira.lbic.fee.unicamp.br:8006/"
  # Set to true if using self-signed certificates on the Proxmox host
  insecure = true

  ssh {
    agent    = false
    username = "root"
    private_key = file("~/.ssh/id_ed25519")

    node {
      name    = "andira"
      address = "andira.lbic.fee.unicamp.br"
    }
  }
}
#+end_src

*** Virtual Machine
This block provisions the VyOS 1.4 router VM. It attaches to the external network (=vmbr1=) and the internal VLAN trunk (=vmbr2=).
**** Resource
#+begin_src tofu
resource "proxmox_virtual_environment_vm" "vyos" {
  name        = "vyos"
  description = "VyOS Core Router"
  node_name   = "andira"
  vm_id       = 201
  tags        = ["core", "router", "vyos"]

  # router must boot before LXCs and Kubernetes
  started = true
  on_boot = true
  startup {
    order      = 1
    up_delay   = 10
  }

  # communicates with the injected qemu-guest-agent
  agent {
    enabled = true
  }
  stop_on_destroy = true
#+end_src

#+begin_src tofu
  cpu {
    cores = 1
    type  = "host"
  }
#+end_src

#+begin_src tofu

    dedicated = 4096
    floating  = 0
  }
#+end_src

***** eth0
#+begin_src tofu
  # eth0: WAN Interface
  # Transit Network (facing the physical network)
  network_device {
    bridge = "vmbr1"

#+end_src

  description = "Kanidm IAM Server"
  tags        = ["core", "iam", "kanidm"]

  vm_id       = 205

  # ensure unprivileged mode

  # autostart

  # enable nesting (recommended for systemd/auth daemons)

    nesting = true

  # ensure the container starts after the network and DNS

    order      = 15
    up_delay   = 5
    down_delay = 5
