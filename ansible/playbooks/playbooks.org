*** Kanidm CLI Authentication
#+begin_src yaml :padline no
- name: ops | kanidm users | authenticate CLI as idm_admin
  ansible.builtin.shell: "KANIDM_PASSWORD=$(cat /root/kanidm_idm_admin_password.txt) kanidm login --name idm_admin"
  changed_when: false
  no_log: true
#+end_src
*** Create User Accounts
#+begin_src yaml
- name: ops | users | create user accounts
  ansible.builtin.command: kanidm person create {{ item.username }} "{{ item.name }}" --name idm_admin
  loop: "{{ users }}"
  register: kanidm_user_create
  changed_when:
    - kanidm_user_create.rc == 0
    - "'already' not in kanidm_user_create.stdout | lower"
    - "'already' not in kanidm_user_create.stderr | lower"
  failed_when:
    - kanidm_user_create.rc != 0
    - "'already exists' not in kanidm_user_create.stderr"
    - "'Entry already exists' not in kanidm_user_create.stderr"
    - "'409 Conflict' not in kanidm_user_create.stderr"
#+end_src

#+begin_src yaml

*** Check POSIX Attributes For Users
#+begin_src yaml
- name: ops | users | inspect POSIX attributes for users
  ansible.builtin.command: kanidm person posix show --name idm_admin {{ item.username }}
  loop: "{{ users }}"
  register: kanidm_posix_state
  changed_when: false
  failed_when: false
#+end_src
*** Give POSIX Attributes Users
#+begin_src yaml
- name: ops | users | give the student POSIX attributes (bash shell)
  ansible.builtin.command: kanidm person posix set {{ item.username }} --shell /bin/bash --name idm_admin
  loop: "{{ users }}"
  loop_control:
    index_var: kanidm_user_index
  when: >-
    kanidm_posix_state.results[kanidm_user_index].rc != 0 or
    'shell: /bin/bash' not in kanidm_posix_state.results[kanidm_user_index].stdout
  register: kanidm_posix
  changed_when: kanidm_posix.rc == 0
#+end_src

*** Add Users to Headscale Group
#+begin_src yaml
- name: ops | users | add users to Headscale VPN group
  ansible.builtin.command: kanidm group add-members headscale_users {{ item.username }} --name idm_admin
  loop: "{{ users }}"
  loop_control:
    index_var: kanidm_user_index
  when:
    - "'memberof: headscale_users@' not in kanidm_person_state.results[kanidm_user_index].stdout"
  register: kanidm_group_add
  changed_when: kanidm_group_add.rc == 0

*** Check POSIX For Workstation Groups

*** Grant POSIX For Workstation Groups
#+begin_src yaml

  loop: "{{ workstations }}"
  loop_control:
    index_var: ws_index

    - ws_users_posix_get.results[ws_index].rc != 0 or 'gidnumber' not in ws_users_posix_get.results[ws_index].stdout | lower
  register: kanidm_ws_users_posix
  changed_when: kanidm_ws_users_posix.rc == 0

- name: ops | workstations | enable POSIX on workstation admins group

  loop: "{{ workstations }}"

*** Enforce Group Memberships (RBAC)
#+begin_src yaml
- name: ops | workstations | set exact members for workstation users
  ansible.builtin.command: "kanidm group set-members {{ item.hostname }}_users {{ item.users_allowed | join(' ') }} --name idm_admin"
  loop: "{{ workstations }}"
  when: item.users_allowed is defined and item.users_allowed | length > 0
  changed_when: true

- name: ops | workstations | set exact members for workstation admins
  ansible.builtin.command: "kanidm group set-members {{ item.hostname }}_admins {{ item.admin | join(' ') }} --name idm_admin"
  loop: "{{ workstations }}"
  when: item.admin is defined and item.admin | length > 0
  changed_when: true

*** Check if Users Exist in IAM
#+begin_src yaml
- name: ops | offboard | check if user exists in Kanidm
  ansible.builtin.command: kanidm person get {{ item.username }} --name idm_admin
  loop: "{{ users_offboard | default([]) }}"
  register: kanidm_user_check
  changed_when: false
  failed_when: false
#+end_src
*** Delete User Accounts from IAM
#+begin_src yaml
- name: ops | offboard | delete user accounts from Kanidm
  ansible.builtin.command: kanidm person delete {{ item.username }} --name idm_admin
  loop: "{{ users_offboard | default([]) }}"
  loop_control:
    index_var: idx
  # idempotency: only runs if the last command succeeded and username exists
  when:
    - kanidm_user_check.results[idx].rc == 0
    - item.username in kanidm_user_check.results[idx].stdout
  register: kanidm_user_delete
  changed_when: kanidm_user_delete.rc == 0
#+end_src
*** Fetch Users from Headscale
#+begin_src yaml
- name: ops | offboard | fetch registered Headscale users
  ansible.builtin.command: headscale users list --output json
  delegate_to: headscale
  register: headscale_users_raw
  changed_when: false
#+end_src
*** Delete Users from Headscale
#+begin_src yaml
- name: ops | offboard | delete user namespace from Headscale
  ansible.builtin.command: "headscale users destroy --name {{ item.username }}"
  loop: "{{ users_offboard | default([]) }}"
  delegate_to: headscale
  # idempotency: parse JSON and runs only if user exists
  when: (headscale_users_raw.stdout | default('[]', true) | from_json | selectattr('name', 'equalto', item.username) | list | length) > 0
  register: headscale_delete
  changed_when: true
#+end_src
*** Delete Home Directories from Workstations
#+begin_src yaml
- name: ops | offboard | delete user home directories on workstations
  ansible.builtin.file:
    path: "/home/{{ item.0.username }}"
    state: absent
  # The 'product' filter creates a nested loop pairing every user with every workstation
  loop: "{{ (users_offboard | default([])) | product(workstations | default([])) | list }}"
  delegate_to: "{{ item.1.hostname }}"
#+end_src
