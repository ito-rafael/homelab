* Headscale
** User Onboarding
:PROPERTIES:
:header-args+: :tangle headscale_onboard_student.yml
:END:
This playbook creates a user in Headscale and generates a Pre-Auth key.
Usage: =ansible-playbook -i inventory.yml onboard_student.yml -e "student=rafael"=
*** Playbook Definition
#+begin_src yaml
---
- name: headscale | onboarding | onboard new VPN user
  hosts: headscale
  gather_facts: false
  vars:
    # fails if "student" is not provided via -e flag
    target_user: "{{ student | mandatory }}"
  tasks:
#+end_src
*** Create User
#+begin_src yaml
    - name: headscale | onboarding | ensure user exists in Headscale
      ansible.builtin.command: "headscale users create {{ target_user }}"
      register: user_creation
      failed_when:
        - user_creation.rc != 0
        - "'already exists' not in user_creation.stderr"
        - "'UNIQUE constraint failed' not in user_creation.stderr"
      changed_when: user_creation.rc == 0
#+end_src
*** Fetch User ID
#+begin_src yaml
    - name: headscale | onboarding | fetch user list to get the numeric ID
      ansible.builtin.command: "headscale users list --output json"
      register: users_json
      changed_when: false
#+end_src
*** Generate Pre-Auth Key
#+begin_src yaml
    - name: headscale | onboarding | generate a 24-hour pre-auth key
      ansible.builtin.command: "headscale preauthkeys create --user {{ (users_json.stdout | from_json | selectattr('name', 'equalto', target_user) | first).id }} --expiration 24h"
      register: preauth_key
      changed_when: true
#+end_src
*** Output Instructions
#+begin_src yaml
    - name: headscale | onboarding | display the connection command for the student
      ansible.builtin.debug:
        msg:
          - "User '{{ target_user }}' is ready."
          - "Send this exact command to the student:"
          - "sudo tailscale up --login-server http://10.10.20.2:8080 --authkey {{ preauth_key.stdout }}"
#+end_src
* VyOS
** Build VyOS Image
:PROPERTIES:
:header-args+: :tangle vyos_build_image.yml
:END:
This playbook natively compiles the VyOS Cloud-Init QCOW2 image directly on the Proxmox host using the official vyos-build Docker environment.
Usage: =ansible-playbook -i inventory/hosts.yml playbooks/build_vyos.yml=
*** Playbook Definition
#+begin_src yaml
---
- name: vyos | build vyos | build VyOS cloud-init image
  hosts: proxmox
  become: yes
  vars:
    vyos_build_dir: /opt/vyos-build
    vyos_target_image: /var/lib/vz/template/iso/vyos-1.4-cloud-init.qcow2
  tasks:
#+end_src
*** Check Existing Image
#+begin_src yaml
    - name: vyos | build vyos | check if VyOS image already exists
      ansible.builtin.stat:
        path: "{{ vyos_target_image }}"
      register: vyos_image_stat
#+end_src
*** Build Block Start
#+begin_src yaml
    - name: vyos | build vyos | build and deploy image
      when: not vyos_image_stat.stat.exists
      block:
#+end_src
*** Install Dependencies
#+begin_src yaml
        - name: vyos | build vyos | install docker and git
          ansible.builtin.apt:
            name:
              - docker.io
              - git
            state: present
            update_cache: yes
#+end_src
*** Clone Repository
#+begin_src yaml
        - name: vyos | build vyos | clone vyos-build repository
          ansible.builtin.git:
            repo: https://github.com/vyos/vyos-build.git
            dest: "{{ vyos_build_dir }}"
            version: current
#+end_src
*** Create Flavor
#+begin_src yaml
        - name: vyos | build vyos | create Proxmox build flavor
          ansible.builtin.copy:
            dest: "{{ vyos_build_dir }}/data/build-flavors/proxmox.toml"
            content: |
              image_format = "qcow2"
              packages = ["qemu-guest-agent"]
            mode: '0644'
#+end_src
*** Compile Image
#+begin_src yaml
        - name: vyos | build vyos | compile VyOS image via Docker
          ansible.builtin.command: >
            docker run --rm --privileged
            -v /dev:/dev
            -v {{ vyos_build_dir }}:/vyos
            -w /vyos
            vyos/vyos-build:current
            bash -c "sudo make clean && sudo ./build-vyos-image proxmox --architecture amd64 --build-by 'ansible@unicamp'"
          args:
            chdir: "{{ vyos_build_dir }}"
#+end_src
*** Find Generated File
#+begin_src yaml
        - name: vyos | build vyos | find the generated QCOW2 image
          ansible.builtin.find:
            paths: "{{ vyos_build_dir }}/build"
            patterns: "vyos-*.qcow2"
          register: built_image
#+end_src
*** Move Image
#+begin_src yaml
        - name: vyos | build vyos | move and rename the VyOS image for OpenTofu
          ansible.builtin.copy:
            src: "{{ built_image.files[0].path }}"
            dest: "{{ vyos_target_image }}"
            remote_src: yes
#+end_src
*** Cleanup
#+begin_src yaml
        - name: vyos | build vyos | clean up the build directory to save space
          ansible.builtin.file:
            path: "{{ vyos_build_dir }}"
            state: absent
#+end_src
