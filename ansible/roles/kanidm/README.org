*** Install Dependencies

- name: core | kanidm setup | install repository and cryptography dependencies

      - curl
      - gnupg
      - apt-transport-https
      - python3-cryptography
    state: present
    update_cache: yes
#+end_src
*** Configure APT Repository
**** Download Kanidm GPG key
#+begin_src yaml
- name: core | kanidm setup | download Kanidm GPG key
  ansible.builtin.get_url:
    url: https://kanidm.github.io/kanidm_ppa/kanidm_ppa.asc
    dest: /etc/apt/trusted.gpg.d/kanidm_ppa.asc
    mode: '0644'
#+end_src
**** Add Kanidm APT repository
#+begin_src yaml
- name: core | kanidm setup | add Kanidm APT repository (deb822 format)
  ansible.builtin.deb822_repository:
    name: kanidm
    types: deb
    architectures: amd64
    uris: https://kanidm.github.io/kanidm_ppa
    suites: "{{ ansible_facts['distribution_release'] | lower }}"
    components: stable
    signed_by: /etc/apt/trusted.gpg.d/kanidm_ppa.asc
    state: present
#+end_src
*** Install Kanidm server and clients
#+begin_src yaml
- name: core | kanidm setup | install Kanidm server and clients
  ansible.builtin.apt:
    name:
      - kanidm   # Kanidm Client Tools
      - kanidmd  # Kanidm Server Daemon
    state: present
    update_cache: yes
#+end_src
*** Generate TLS Certificates
Kanidm enforces strict TLS. Since we are using a self-signed chain, we generate it locally on the LXC container.
**** Create config directory
#+begin_src yaml
- name: core | kanidm setup | create kanidm config directory
  ansible.builtin.file:
    path: /etc/kanidm
    state: directory
    owner: root
    group: kanidmd
    mode: '0770'
#+end_src
**** Generate private key
#+begin_src yaml
- name: core | kanidm setup | generate private key for TLS
  community.crypto.openssl_privatekey:
    path: /etc/kanidm/key.pem
    type: RSA
    size: 2048
    format: pkcs8
    owner: root
    group: kanidmd
    mode: '0640'
#+end_src
**** Generate CSR
#+begin_src yaml
- name: core | kanidm setup | generate CSR for Kanidm
  community.crypto.openssl_csr:
    path: /etc/kanidm/server.csr
    privatekey_path: /etc/kanidm/key.pem
    common_name: "{{ kanidm_domain }}"
    subject_alt_name:
      - "DNS:kanidm"
      - "DNS:{{ kanidm_domain }}"
      - "IP:{{ kanidm_ip }}"
#+end_src
**** Generate self-signed certificate
#+begin_src yaml
- name: core | kanidm setup | generate self-signed TLS certificate
  community.crypto.x509_certificate:
    path: /etc/kanidm/chain.pem
    privatekey_path: /etc/kanidm/key.pem
    csr_path: /etc/kanidm/server.csr
    provider: selfsigned
    owner: root
    group: kanidmd
    mode: '0640'
#+end_src

*** Recover Default admin Account
Kanidm separates global configuration (admin) from user management (idm_admin).
**** Check if admin password was recovered
#+begin_src yaml
- name: core | kanidm setup | check if admin password was already recovered
  ansible.builtin.stat:
    path: /etc/kanidm/.admin_recovered
  register: admin_recovered
#+end_src
**** Recover system admin password
#+begin_src yaml
- name: core | kanidm setup | recover system admin password
  ansible.builtin.command: kanidmd recover-account -c /etc/kanidm/server.toml admin
  register: kanidm_sys_admin_recovery
  when: not admin_recovered.stat.exists
#+end_src
**** Save system admin password
#+begin_src yaml
- name: core | kanidm setup | save system admin password to root home directory
  ansible.builtin.copy:
    content: >-
      {{ (kanidm_sys_admin_recovery.stdout | split('new_password: "'))[1] | replace('"', '') | trim }}
    dest: /root/kanidm_admin_password.txt
    mode: '0600'
  when: not admin_recovered.stat.exists
#+end_src
**** Mark admin password as recovered
#+begin_src yaml
- name: core | kanidm setup | mark admin password as recovered
  ansible.builtin.file:
    path: /etc/kanidm/.admin_recovered
    state: touch
  when: not admin_recovered.stat.exists
  no_log: true
#+end_src
**** Authenticate CLI as admin
#+begin_src yaml
- name: core | kanidm setup | authenticate CLI as system admin
  ansible.builtin.shell: "KANIDM_PASSWORD=$(cat /root/kanidm_admin_password.txt) kanidm login --name admin"
  changed_when: false
#+end_src

*** Check POSIX Attributes For Users
#+begin_src yaml

  changed_when: false
  failed_when: false
#+end_src
*** Give POSIX Attributes Users
#+begin_src yaml
- name: core | kanidm headscale | give the student POSIX attributes (bash shell)
  ansible.builtin.command: kanidm person posix set {{ item.username }} --shell /bin/bash --name idm_admin
  loop: "{{ users }}"
  loop_control:
    index_var: kanidm_user_index
  when: >-
    kanidm_posix_state.results[kanidm_user_index].rc != 0 or
    'shell: /bin/bash' not in kanidm_posix_state.results[kanidm_user_index].stdout
  register: kanidm_posix
  changed_when: kanidm_posix.rc == 0
#+end_src
