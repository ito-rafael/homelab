*** Install Dependencies
#+begin_src yaml :padline no
- name: core | kanidm setup | install repository and cryptography dependencies
  ansible.builtin.apt:
    name:
      - curl
      - gnupg
      - apt-transport-https
      - python3-cryptography
    state: present
    update_cache: yes
#+end_src
*** Configure APT Repository
**** Download Kanidm GPG key
#+begin_src yaml
- name: core | kanidm setup | download Kanidm GPG key
  ansible.builtin.get_url:
    url: https://kanidm.github.io/kanidm_ppa/kanidm_ppa.asc
    dest: /etc/apt/trusted.gpg.d/kanidm_ppa.asc
    mode: '0644'
#+end_src
**** Add Kanidm APT repository
#+begin_src yaml
- name: core | kanidm setup | add Kanidm APT repository (deb822 format)
  ansible.builtin.deb822_repository:
    name: kanidm
    types: deb
    architectures: amd64
    uris: https://kanidm.github.io/kanidm_ppa
    suites: "{{ ansible_facts['distribution_release'] | lower }}"
    components: stable
    signed_by: /etc/apt/trusted.gpg.d/kanidm_ppa.asc
    state: present
#+end_src
*** Install Kanidm server and clients
#+begin_src yaml
- name: core | kanidm setup | install Kanidm server and clients
  ansible.builtin.apt:
    name:
      - kanidm   # Kanidm Client Tools
      - kanidmd  # Kanidm Server Daemon
    state: present
    update_cache: yes
#+end_src
*** Generate TLS Certificates
Kanidm enforces strict TLS. Since we are using a self-signed chain, we generate it locally on the LXC container.
**** Create config directory
#+begin_src yaml
- name: core | kanidm setup | create kanidm config directory
  ansible.builtin.file:
    path: /etc/kanidm
    state: directory
    owner: root
    group: kanidmd
    mode: '0770'
#+end_src
**** Generate private key
#+begin_src yaml
- name: core | kanidm setup | generate private key for TLS
  community.crypto.openssl_privatekey:
    path: /etc/kanidm/key.pem
    owner: root
    group: kanidmd
    mode: '0640'
#+end_src
**** Generate CSR
#+begin_src yaml
- name: core | kanidm setup | generate CSR for Kanidm
  community.crypto.openssl_csr:
    path: /etc/kanidm/server.csr
    privatekey_path: /etc/kanidm/key.pem
    common_name: "{{ kanidm_domain }}"
    subject_alt_name:
      - "DNS:kanidm"
      - "DNS:{{ kanidm_domain }}"
      - "IP:{{ kanidm_ip }}"
#+end_src
**** Generate self-signed certificate
#+begin_src yaml
- name: core | kanidm setup | generate self-signed TLS certificate
  community.crypto.x509_certificate:
    path: /etc/kanidm/chain.pem
    privatekey_path: /etc/kanidm/key.pem
    csr_path: /etc/kanidm/server.csr
    provider: selfsigned
    owner: root
    group: kanidmd
    mode: '0640'
#+end_src
*** Configure and Start Daemon
**** Create database directory
#+begin_src yaml
- name: core | kanidm setup | create kanidmd database directory
  ansible.builtin.file:
    path: /var/lib/kanidmd
    state: directory
    owner: root
    group: kanidmd
    mode: '0750'
#+end_src
**** Deploy server.toml configuration
#+begin_src yaml
- name: core | kanidm setup | deploy server.toml configuration
  ansible.builtin.template:
    src: server.toml.j2
    dest: /etc/kanidm/server.toml
    owner: root
    group: kanidmd
    mode: '0640'
  notify: core | kanidm setup | restart systemd service
#+end_src
