** Install & Configure
:PROPERTIES:
:header-args+: :tangle headscale.yml
:END:
#+begin_src yaml
---
#+end_src
*** Download
#+begin_src yaml :padline no
- name: Download Headscale Debian package
  ansible.builtin.get_url:
    url: "https://github.com/juanfont/headscale/releases/download/v{{ headscale_version }}/headscale_{{ headscale_version }}_linux_amd64.deb"
    dest: "/tmp/headscale.deb"
    mode: '0644'
#+end_src
*** Install
#+begin_src yaml
- name: Install Headscale package
  ansible.builtin.apt:
    deb: "/tmp/headscale.deb"
    state: present
#+end_src
*** Create dir
#+begin_src yaml
- name: Ensure Headscale configuration directory exists
  ansible.builtin.file:
    path: /etc/headscale
    state: directory
    mode: '0755'
#+end_src
