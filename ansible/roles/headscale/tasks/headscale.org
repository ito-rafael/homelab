* Headscale
** Install & Configure
:PROPERTIES:
:header-args+: :tangle headscale.yml
:END:
#+begin_src yaml
---
#+end_src
*** Download
#+begin_src yaml :padline no
- name: core | headscale | download Debian package
  ansible.builtin.get_url:
    url: "https://github.com/juanfont/headscale/releases/download/v{{ headscale_version }}/headscale_{{ headscale_version }}_linux_{{ headscale_arch }}.deb"
    dest: "/tmp/headscale.deb"
    mode: '0644'
#+end_src
*** Install
#+begin_src yaml
- name: core | headscale | install Headscale package
  ansible.builtin.apt:
    deb: "/tmp/headscale.deb"
    state: present
#+end_src
*** Create directory
#+begin_src yaml
- name: core | headscale | create Headscale directory
  ansible.builtin.file:
    path: /etc/headscale
    state: directory
    mode: '0755'
#+end_src
*** Configure
#+begin_src yaml
- name: core | headscale | deploy Headscale config.yaml
  ansible.builtin.copy:
    dest: /etc/headscale/config.yaml
    src: config.yaml
    owner: headscale
    group: headscale
    mode: '0644'
  notify: core | headscale | restart systemd service
#+end_src
*** Start service
#+begin_src yaml
- name: core | headscale | enable and start systemd service
  ansible.builtin.systemd:
    name: headscale
    state: started
    enabled: true
#+end_src
** TUN Device Configuration
:PROPERTIES:
:header-args+: :tangle lxc_tun.yml
:END:
#+begin_src yaml
---
#+end_src
*** Grant Access
#+begin_src yaml :padline no
- name: core | proxmox | grant TUN device access to Headscale LXC
  ansible.builtin.lineinfile:
    path: /etc/pve/lxc/202.conf
    line: "{{ item }}"
  loop:
    - "lxc.cgroup2.devices.allow: c 10:200 rwm"
    - "lxc.mount.entry: /dev/net/tun dev/net/tun none bind,create=file"
  delegate_to: proxmox
  register: headscale_tun_config
#+end_src
*** Reboot Container
#+begin_src yaml
- name: core | proxmox | reboot Headscale LXC to apply TUN permissions
  ansible.builtin.shell: pct stop 202 && pct start 202
  delegate_to: proxmox
  when: headscale_tun_config.changed
#+end_src
