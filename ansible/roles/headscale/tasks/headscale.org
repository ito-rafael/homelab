* Headscale
** Install & Configure
:PROPERTIES:
:header-args+: :tangle headscale.yml
:END:
#+begin_src yaml
---
#+end_src
*** Download
#+begin_src yaml :padline no
- name: core | headscale | download Debian package
  ansible.builtin.get_url:
    url: "https://github.com/juanfont/headscale/releases/download/v{{ headscale_version }}/headscale_{{ headscale_version }}_linux_{{ headscale_arch }}.deb"
    dest: "/tmp/headscale.deb"
    mode: '0644'
#+end_src
*** Install
#+begin_src yaml
- name: core | headscale | install Headscale package
  ansible.builtin.apt:
    deb: "/tmp/headscale.deb"
    state: present
#+end_src
*** Create directory
#+begin_src yaml
- name: core | headscale | create Headscale directory
  ansible.builtin.file:
    path: /etc/headscale
    state: directory
    mode: '0755'
#+end_src
*** Create Let's Encrypt directory
#+begin_src yaml
- name: core | headscale | create Let's Encrypt cache directory
  ansible.builtin.file:
    path: /var/lib/headscale/cache
    state: directory
    owner: headscale
    group: headscale
    mode: '0750'
  notify: core | headscale | restart systemd service
#+end_src
*** Configure
#+begin_src yaml
- name: core | headscale | deploy Headscale config.yaml
  ansible.builtin.copy:
    dest: /etc/headscale/config.yaml
    src: config.yaml
    owner: headscale
    group: headscale
    mode: '0644'
  notify: core | headscale | restart systemd service
#+end_src
*** Start service
#+begin_src yaml
- name: core | headscale | enable and start systemd service
  ansible.builtin.systemd:
    name: headscale
    state: started
    enabled: true
#+end_src
** TUN Device Configuration
:PROPERTIES:
:header-args+: :tangle lxc_tun.yml
:END:
#+begin_src yaml
---
#+end_src
*** Get Container ID
#+begin_src yaml :padline no
- name: core | proxmox | get Headscale LXC ID
  ansible.builtin.shell: "pct list | grep -w '{{ inventory_hostname }}' | awk '{print $1}'"
  delegate_to: proxmox
  register: headscale_lxc_id
  changed_when: false
  failed_when: headscale_lxc_id.stdout | trim == ""
#+end_src
*** Grant Access
#+begin_src yaml
- name: core | proxmox | grant TUN device access to Headscale LXC
  ansible.builtin.lineinfile:
    path: "/etc/pve/lxc/{{ headscale_lxc_id.stdout | trim }}.conf"
    line: "{{ item }}"
  loop:
    - "lxc.cgroup2.devices.allow: c 10:200 rwm"
    - "lxc.mount.entry: /dev/net/tun dev/net/tun none bind,create=file"
  delegate_to: proxmox
  register: headscale_tun_config
#+end_src
*** Reboot Container
#+begin_src yaml
- name: core | proxmox | reboot Headscale LXC to apply TUN permissions
  ansible.builtin.shell: "pct stop {{ headscale_lxc_id.stdout | trim }} && pct start {{ headscale_lxc_id.stdout | trim }}"
  delegate_to: proxmox
  when: headscale_tun_config.changed
#+end_src
*** Wait for Reconnect
#+begin_src yaml
- name: core | headscale | wait for Headscale LXC to come back online
  ansible.builtin.wait_for_connection:
    timeout: 60
  when: headscale_tun_config.changed
#+end_src
** Subnet router (Tailscale client)
:PROPERTIES:
:header-args+: :tangle tailscale.yml

*** Enable IP forwarding
#+begin_src yaml
- name: core | headscale | enable IPv4 forwarding for subnet routing
  ansible.posix.sysctl:
    name: net.ipv4.ip_forward
    value: '1'
    sysctl_set: yes
    state: present
    reload: yes
#+end_src
*** Install Tailscale
**** Check if Tailscale is installed
#+begin_src yaml
- name: core | headscale | check if Tailscale is installed
  ansible.builtin.command: which tailscale
  register: tailscale_installed
  changed_when: false
  failed_when: false
#+end_src
**** Install dependencies
#+begin_src yaml
- name: core | headscale | ensure curl and ca-certificates are installed
  ansible.builtin.apt:
    name:
      - curl
      - ca-certificates
    state: present
    update_cache: yes
  when: tailscale_installed.rc != 0
#+end_src
**** Download Tailscale installer
#+begin_src yaml
- name: core | headscale | download Tailscale install script
  ansible.builtin.get_url:
    url: https://tailscale.com/install.sh
    dest: /tmp/install_tailscale.sh
    mode: '0700'
  when: tailscale_installed.rc != 0
#+end_src
**** Install Tailscale
#+begin_src yaml
- name: core | headscale | execute Tailscale installer
  ansible.builtin.command: /tmp/install_tailscale.sh
  when: tailscale_installed.rc != 0
#+end_src
*** Start Tailscale service
#+begin_src yaml
- name: core | headscale | start & enable Tailscale service
  ansible.builtin.systemd:
    name: tailscaled
    state: started
    enabled: true
#+end_src
*** Create User
#+begin_src yaml
- name: core | headscale | create lbic user namespace
  ansible.builtin.command: headscale users create lbic
  register: headscale_user
  failed_when:
    - headscale_user.rc != 0
    - "'UNIQUE constraint failed' not in headscale_user.stderr"
    - "'already exists' not in headscale_user.stderr"
  changed_when: headscale_user.rc == 0
#+end_src

  register: tailscale_status
  changed_when: false
  failed_when: false
#+end_src

**** Get users list
#+begin_src yaml
  - name: core | headscale | get users list as JSON
    ansible.builtin.command: headscale users list --output json
    register: headscale_users_raw
    changed_when: false
#+end_src
**** Extract user ID for lbic
#+begin_src yaml
  - name: core | headscale | extract user ID for lbic
    ansible.builtin.set_fact:
      headscale_user_id: "{{ (headscale_users_raw.stdout | from_json | selectattr('name', 'equalto', 'lbic') | first).id }}"
#+end_src
**** Generate pre-auth key
#+begin_src yaml
  - name: core | headscale | generate pre-auth key
    ansible.builtin.command: "headscale preauthkeys create -e 10m --user {{ headscale_user_id }}"
    register: headscale_authkey
    changed_when: false
#+end_src
**** Advertise the lab subnet
#+begin_src yaml
  - name: core | headscale | start Tailscale and advertise routes
    ansible.builtin.command: >
      tailscale up
      --login-server https://andira.lbic.fee.unicamp.br
      --advertise-routes=143.106.45.160/27
      --authkey={{ headscale_authkey.stdout | trim }}
#+end_src

**** Get nodes list

- name: core | headscale | get nodes list as JSON

  register: headscale_nodes_raw
  changed_when: false
#+end_src
**** Extract node ID for headscale container
#+begin_src yaml
- name: core | headscale | extract node ID for headscale container
  ansible.builtin.set_fact:

#+end_src
**** Approve Routes
#+begin_src yaml
- name: core | headscale | approve advertised routes
  ansible.builtin.command: "headscale nodes approve-routes --identifier {{ headscale_node_id }} --routes 143.106.45.160/27"
  changed_when: false
#+end_src
