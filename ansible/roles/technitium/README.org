*** Install Prerequisites
#+begin_src yaml :padline no
- name: core | dns | install prerequisites
  ansible.builtin.apt:
    name:
      - curl
      - ca-certificates
      - sudo
    state: present
    update_cache: yes
#+end_src
*** Check if Technitium is installed
#+begin_src yaml
- name: core | dns | check if Technitium is installed
  ansible.builtin.stat:
    path: /etc/systemd/system/dns.service
  register: technitium_service
#+end_src
*** Install
#+begin_src yaml
- name: core | dns | install Technitium
  ansible.builtin.shell: "curl -sSL https://download.technitium.com/dns/install.sh | sudo bash"
  when: not technitium_service.stat.exists
#+end_src
*** Start systemd service
#+begin_src yaml
- name: core | dns | start & enable systemd service
  ansible.builtin.systemd:
    name: dns
    state: started
    enabled: yes
    daemon_reload: yes
#+end_src

*** Attempt Secure Login
#+begin_src yaml :padline no
- name: core | dns | attempt login with secure password
  ansible.builtin.uri:
    url: "http://localhost:5380/api/user/login"
    method: POST
    body_format: form-urlencoded
    body:
      user: "admin"
      pass: "{{ technitium_admin_password }}"
  register: login_secure
  failed_when: false
  changed_when: false
#+end_src
*** Fallback to Default Password
#+begin_src yaml
- name: core | dns | fallback to default password (first run only)
  ansible.builtin.uri:
    url: "http://localhost:5380/api/user/login"
    method: POST
    body_format: form-urlencoded
    body:
      user: "admin"
      pass: "admin"
  register: login_default
  when: login_secure.json.get('status') != 'ok'
  changed_when: false
#+end_src
*** Verify Login Success
#+begin_src yaml
- name: core | dns | verify successful login
  ansible.builtin.fail:
    msg: "Both logins failed! Check your vault.yml password."
  when:
    - login_secure.json.get('status') != 'ok'
    - login_default.json.get('status') != 'ok'
#+end_src
*** Set Active Token Fact
#+begin_src yaml
- name: core | dns | set active token fact
  ansible.builtin.set_fact:
    active_token: "{{ login_secure.json.get('token') if login_secure.json.get('status') == 'ok' else login_default.json.get('token') }}"
#+end_src
*** Secure Admin Account
#+begin_src yaml
- name: core | dns | secure the server by changing default password
  ansible.builtin.uri:
    url: "http://localhost:5380/api/user/changePassword"
    method: POST
    body_format: form-urlencoded
    body:
      token: "{{ active_token }}"
      pass: "admin"                               # current password
      newPass: "{{ technitium_admin_password }}"  # new password
  when: login_secure.json.get('status') != 'ok'
  register: password_change
  failed_when: false
  changed_when: password_change.json.get('status') == 'ok'
#+end_src
*** Configure Upstream Forwarders
#+begin_src yaml
- name: core | dns | configure upstream forwarders
  ansible.builtin.uri:
    url: "http://localhost:5380/api/settings/set"
    method: POST
    body_format: form-urlencoded
    body:
      token: "{{ active_token }}"
      forwarders: "143.106.8.30,143.106.2.5,1.1.1.1,1.0.0.1,8.8.8.8,8.8.4.4"
    status_code: 200
  # API does not tell if forwarders were different
  changed_when: false
#+end_src
*** Create Local Authoritative Zone
#+begin_src yaml
- name: core | dns | create local authoritative zone
  ansible.builtin.uri:
    url: "http://localhost:5380/api/zones/create"
    method: POST
    body_format: form-urlencoded
    body:
      token: "{{ active_token }}"
      domain: "internal"
      type: "Primary"
    status_code: 200
  register: zone_result
  failed_when: false
  changed_when: zone_result.json.get('status') == 'ok'
#+end_src
*** Add Infrastructure A Records
#+begin_src yaml
- name: core | dns | add A records for core infrastructure
  ansible.builtin.uri:
    url: "http://localhost:5380/api/zones/records/add"
    method: POST
    body_format: form-urlencoded
    body:
      token: "{{ active_token }}"
      domain: "{{ item.name }}.internal"
      type: "A"
      ipAddress: "{{ item.ip }}"
    status_code: 200
  register: record_results
  failed_when: false
  changed_when: record_results.json.get('status') == 'ok'
  loop:
    - { name: 'proxmox',   ip: '10.10.10.254' }
    - { name: 'vyos',      ip: '10.10.10.1' }
    - { name: 'headscale', ip: '10.10.20.2' }
    - { name: 'dns-1',     ip: '10.10.20.3' }
    - { name: 'dns-2',     ip: '10.10.20.4' }
#+end_src
