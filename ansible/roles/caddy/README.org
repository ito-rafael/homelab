*** Configure APT Repository
**** Download Caddy GPG key
#+begin_src yaml :tangle no
- name: core | caddy | download Caddy GPG key
  ansible.builtin.get_url:
    url: https://dl.cloudsmith.io/public/caddy/stable/gpg.key
    dest: /usr/share/keyrings/caddy-stable-archive-keyring.asc
    mode: '0644'
#+end_src
**** Add Caddy APT repository
#+begin_src yaml :tangle no
- name: core | caddy | add Caddy APT repository (deb822 format)
  ansible.builtin.deb822_repository:
    name: caddy-stable
    types: deb
    uris: https://dl.cloudsmith.io/public/caddy/stable/deb/debian
    suites: any-version
    components: main
    signed_by: /usr/share/keyrings/caddy-stable-archive-keyring.asc
    state: present
#+end_src
*** Install Caddy web server
#+begin_src yaml :tangle no
- name: core | caddy | install Caddy
  ansible.builtin.apt:
    name: caddy
    state: present
    update_cache: yes
#+end_src
*** Install Caddy Web Server From GitHub Releases
**** Get Latest Release Version
#+begin_src yaml
- name: core | caddy | get latest Caddy release metadata
  ansible.builtin.uri:
    url: https://api.github.com/repos/caddyserver/caddy/releases/latest
    return_content: true
  register: caddy_latest
  changed_when: false
#+end_src
**** Download Latest Debian Package
#+begin_src yaml
- name: core | caddy | download Caddy Debian package
  ansible.builtin.get_url:
    url: "https://github.com/caddyserver/caddy/releases/download/{{ caddy_latest.json.tag_name }}/caddy_{{ caddy_latest.json.tag_name | regex_replace('^v', '') }}_linux_amd64.deb"
    dest: /tmp/caddy.deb
    mode: '0644'
#+end_src
*** Create Let's Encrypt Directory
#+begin_src yaml
- name: core | caddy | create Let's Encrypt ACME directory
  ansible.builtin.file:
    path: /var/lib/caddy/.local/share/caddy
    state: directory
    mode: '0755'
#+end_src
*** Fix Let's Encrypt Directory Permissions
#+begin_src yaml
- name: core | caddy | set recursive ownership for Caddy state dir
  ansible.builtin.file:
    path: /var/lib/caddy
    state: directory
    owner: caddy
    group: caddy
    recurse: yes
  notify: core | caddy | restart systemd service
#+end_src
*** Configure and Start Daemon
**** Deploy Caddyfile configuration
#+begin_src yaml
- name: core | caddy | deploy Caddyfile
  ansible.builtin.template:
    src: Caddyfile.j2
    dest: /etc/caddy/Caddyfile
    owner: root
    group: root
    mode: '0644'
  notify: core | caddy | restart systemd service
#+end_src
**** Enable and start service
#+begin_src yaml
- name: core | caddy | enable and start caddy service
  ansible.builtin.systemd:
    name: caddy
    state: started
    enabled: yes
    daemon_reload: true
#+end_src

**** Kanidm
#+begin_src conf
  # route Kanidm IAM Traffic
  @kanidm path /ui* /oauth2* /v1* /rs* /status* /pkg* /icons* /favicon.ico
    handle @kanidm {
      # Kanidm uses HTTPS strictly. We tell Caddy to connect via HTTPS
      # but ignore the backend's self-signed certificate.
      reverse_proxy https://10.10.20.5:443 {
        transport http {
            tls_insecure_skip_verify
        }
      }
    }
#+end_src
