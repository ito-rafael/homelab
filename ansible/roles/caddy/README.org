*** Install Dependencies
#+begin_src yaml :padline no
- name: core | caddy | install repository dependencies
  ansible.builtin.apt:
    name:
      - debian-keyring
      - debian-archive-keyring
      - apt-transport-https
      - curl
    state: present
    update_cache: yes
#+end_src
*** Configure APT Repository
**** Download Caddy GPG key
#+begin_src yaml
- name: core | caddy | download Caddy GPG key
  ansible.builtin.get_url:
    url: https://dl.cloudsmith.io/public/caddy/stable/gpg.key
    dest: /usr/share/keyrings/caddy-stable-archive-keyring.asc
    mode: '0644'
#+end_src
**** Add Caddy APT repository
#+begin_src yaml
- name: core | caddy | add Caddy APT repository (deb822 format)
  ansible.builtin.deb822_repository:
    name: caddy-stable
    types: deb
    uris: https://dl.cloudsmith.io/public/caddy/stable/deb/debian
    suites: any-version
    components: main
    signed_by: /usr/share/keyrings/caddy-stable-archive-keyring.asc
    state: present
#+end_src
*** Install Caddy web server
#+begin_src yaml
- name: core | caddy | install Caddy
  ansible.builtin.apt:
    name: caddy
    state: present
    update_cache: yes
#+end_src
*** Configure and Start Daemon
**** Deploy Caddyfile configuration
#+begin_src yaml
- name: core | caddy | deploy Caddyfile
  ansible.builtin.template:
    src: Caddyfile.j2
    dest: /etc/caddy/Caddyfile
    owner: root
    group: root
    mode: '0644'
  notify: core | caddy | restart systemd service
#+end_src
**** Enable and start service
#+begin_src yaml
- name: core | caddy | enable and start caddy service
  ansible.builtin.systemd:
    name: caddy
    state: started
    enabled: yes
    daemon_reload: true
#+end_src

* Templates
** Caddyfile.j2
#+begin_src conf :tangle templates/Caddyfile.j2
# global options
{
  email {{ vault_smtp_user }}
}

# Kanidm: port 443
{{ caddy_domain }} {
  # Kanidm uses HTTPS strictly. We tell Caddy to connect via HTTPS
  # but ignore the backend's self-signed certificate.
  reverse_proxy https://{{ caddy_kanidm_ip }}:443 {
    transport http {
      tls_insecure_skip_verify
    }
  }
}

# Headscale: port 8443
{{ caddy_domain }}:8443 {
  # Headscale listens locally on plain HTTP, Caddy wraps it in HTTPS
  reverse_proxy http://{{ caddy_headscale_ip }}:8080 {
    # force Caddy to preserve the original 8443 port in the Host header
    # so Headscale's internal redirects do not default to 443
    header_up Host {host}:{port}
  }
}
#+end_src
