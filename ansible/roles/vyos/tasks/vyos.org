* VyOS
** Build Image
:PROPERTIES:
:header-args+: :tangle build_image.yml
:END:
These tasks build a reproducible VyOS Rolling QCOW2 image.

The VyOS source tree is pinned to an exact commit rather than following the
moving =rolling= branch. The build container is built locally from the same
pinned =vyos-build= checkout, keeping the builder environment aligned with the
source revision.

The resulting QCOW2 image includes =qemu-guest-agent= for Proxmox integration.
The image is built with Cloud-Init, but further configured by Ansible after
deployment using the router configuration tasks.

The build also performs a preflight check against the current VyOS Rolling
package repository. The kernel version and flavor are read from the pinned
=vyos-build= source tree and the corresponding kernel package is verified
before the expensive image build starts. This prevents failures caused by a
pinned source revision referencing a package that is no longer available in
the Rolling repository.

After the build completes, the generated QCOW2 image is validated with
=qemu-img= before being copied to the Proxmox template directory.

*** Install Dependencies
#+begin_src yaml
    - name: vyos | build image | install dependencies
      ansible.builtin.apt:

          - podman
          - git
          - genisoimage
        state: present

*** Checkout Pinned Repository
#+begin_src yaml
    - name: vyos | build image | checkout pinned vyos-build repository
      ansible.builtin.git:
        repo: "{{ vyos_build_repo }}"
        dest: "{{ vyos_build_dir }}"
        version: "{{ vyos_build_ref }}"
        force: true
        update: true
#+end_src

**** Build NoCloud Seed ISO
#+begin_src yaml
    - name: vyos | build image | create NoCloud seed ISO
      ansible.builtin.command:
        cmd: >
          genisoimage
          -joliet
          -rock
          -volid cidata
          -output {{ vyos_seed_iso }}
          meta-data
          user-data
          network-config
      args:
        chdir: "{{ vyos_seed_dir }}"
#+end_src
**** Validate NoCloud Seed ISO
#+begin_src yaml
    - name: vyos | build image | validate NoCloud seed ISO
      ansible.builtin.command:

      register: vyos_seed_iso_info
      changed_when: false
#+end_src
**** Assert NoCloud Seed ISO
#+begin_src yaml
    - name: vyos | build image | verify NoCloud seed ISO label

          NoCloud seed ISO does not have the expected cidata volume label.
#+end_src

*** Verify Checked-Out Commit
#+begin_src yaml
    - name: vyos | build image | read checked-out commit
      ansible.builtin.command:
        cmd: git rev-parse HEAD
        chdir: "{{ vyos_build_dir }}"
      register: vyos_actual_commit
      changed_when: false
#+end_src
*** Assert Pinned Commit
#+begin_src yaml
    - name: vyos | build image | verify checked-out commit
      ansible.builtin.assert:
        that:
          - vyos_actual_commit.stdout | trim == vyos_build_ref
        fail_msg: >-
          vyos-build checkout is not at the requested commit.
          Expected {{ vyos_build_ref }},
          got {{ vyos_actual_commit.stdout | trim }}.
#+end_src

*** Check Builder Image
#+begin_src yaml
    - name: vyos | build image | check if pinned builder image exists
      ansible.builtin.command:
        cmd: "podman image exists {{ vyos_builder_image }}"
      register: vyos_builder_exists
      changed_when: false
      failed_when: false
#+end_src
*** Build Pinned Builder Container
#+begin_src yaml
    - name: vyos | build image | build pinned VyOS builder container
      ansible.builtin.command:
        cmd: >
          podman build
          --tag {{ vyos_builder_image }}
          {{ vyos_build_dir }}/docker
      args:
        chdir: "{{ vyos_build_dir }}"
      when: vyos_builder_exists.rc != 0
#+end_src
*** Read Kernel Version
#+begin_src yaml
    - name: vyos | build image | read kernel version from pinned source
      ansible.builtin.shell:
        cmd: >
          awk -F'"' '/^[[:space:]]*kernel_version[[:space:]]*=/{print $2}'
          data/defaults.toml
        chdir: "{{ vyos_build_dir }}"
      register: vyos_kernel_version
      changed_when: false
#+end_src
*** Read Kernel Flavor
#+begin_src yaml
    - name: vyos | build image | read kernel flavor from pinned source
      ansible.builtin.shell:
        cmd: >
          awk -F'"' '/^[[:space:]]*kernel_flavor[[:space:]]*=/{print $2}'
          data/defaults.toml
        chdir: "{{ vyos_build_dir }}"
      register: vyos_kernel_flavor
      changed_when: false
#+end_src
*** Validate Kernel Metadata
#+begin_src yaml
    - name: vyos | build image | validate kernel metadata
      ansible.builtin.assert:
        that:
          - vyos_kernel_version.stdout | trim | length > 0
          - vyos_kernel_flavor.stdout | trim | length > 0
        fail_msg: >-
          Could not determine kernel_version and/or kernel_flavor from
          {{ vyos_build_dir }}/data/defaults.toml.
#+end_src
*** Define Kernel Package
#+begin_src yaml
    - name: vyos | build image | define required kernel package
      ansible.builtin.set_fact:
        vyos_kernel_package: >-
          linux-image-{{ vyos_kernel_version.stdout | trim }}-{{ vyos_kernel_flavor.stdout | trim }}
#+end_src
*** Check Kernel Package Availability
#+begin_src yaml
    - name: vyos | build image | check required kernel package availability
      ansible.builtin.command:
        cmd: >
          podman run
          --rm
          {{ vyos_builder_image }}
          bash -lc
          "apt-get update -qq &&
           apt-cache policy {{ vyos_kernel_package }}"
      args:
        chdir: "{{ vyos_build_dir }}"
      register: vyos_kernel_package_check
      changed_when: false
      failed_when: false
#+end_src
*** Assert Kernel Package Availability
#+begin_src yaml
    - name: vyos | build image | verify required kernel package is available
      ansible.builtin.assert:
        that:
          - vyos_kernel_package_check.rc == 0
        fail_msg: |
          The pinned vyos-build revision {{ vyos_build_ref }} requires
          {{ vyos_kernel_package }}, but that package is not currently
          available from the configured VyOS Rolling package repository.

          Kernel version:
          {{ vyos_kernel_version.stdout | trim }}
          Kernel flavor:
          {{ vyos_kernel_flavor.stdout | trim }}

          This usually means that the pinned vyos-build revision and the
          current Rolling package repositories are out of sync. Select a
          newer known-good vyos-build commit or wait until the corresponding
          package is published.
#+end_src

*** Compile and Build Image
#+begin_src yaml
    - name: vyos | build image | compile VyOS QCOW2 image
      ansible.builtin.command:
        cmd: >
          podman run
          --rm
          --privileged
          --sysctl net.ipv6.conf.lo.disable_ipv6=0
          -v {{ vyos_build_dir }}:/vyos
          -w /vyos
          {{ vyos_builder_image }}
          bash -lc
          "sudo ./build-vyos-image
          --architecture {{ vyos_architecture }}
          --build-by '{{ vyos_build_by }}'
          --custom-package cloud-init
          {{ vyos_build_flavor }}"
      args:
        chdir: "{{ vyos_build_dir }}"
#+end_src

*** Cleanup
The build directory is intentionally retained so build logs and the exact
checked-out source remain available for debugging and inspection.

It can be removed manually after the image has been validated if disk space is more
important than retaining the build environment.
#+begin_src yaml :tangle no
- name: vyos | build image | clean up the build directory
  ansible.builtin.file:
    path: "{{ vyos_build_dir }}"
    state: absent

*** Configure router
#+begin_src yaml
- name: vyos | routing | configure interfaces, DNS, and NAT
  vyos.vyos.vyos_config:
    lines:
      - set interfaces ethernet eth0 description 'WAN-TRANSIT'
      - set interfaces ethernet eth1 description 'LAN-TRUNK'
      - set interfaces ethernet eth1 vif 20 address '10.10.20.1/24'
      - set interfaces ethernet eth1 vif 20 description 'INFRA-VLAN'
      - set system name-server '1.1.1.1'
      - set protocols static route 0.0.0.0/0 next-hop '10.10.10.254'
      - set nat source rule 100 outbound-interface name 'eth0'
      - set nat source rule 100 source address '10.10.0.0/16'
      - set nat source rule 100 translation address 'masquerade'
  register: vyos_config_result
#+end_src
*** Save configuration
#+begin_src yaml
- name: vyos | routing | save configuration
  vyos.vyos.vyos_command:
    commands:
      - configure
      - save
      - exit
  changed_when: true
  when: vyos_config_result.changed
