*** Kanidm authentication
:PROPERTIES:
:header-args+: :tangle tasks/system/kanidm-unixd.yml
:END:
#+begin_src yaml
---
#+end_src
**** Install APT Transport
#+begin_src yaml :padline no
- name: system | kanidm | install APT transport and dependencies
  ansible.builtin.apt:
    name: apt-transport-https
    state: present
    update_cache: yes
#+end_src
**** Configure APT Repository
***** Download Kanidm GPG key
#+begin_src yaml
- name: system | kanidm | download Kanidm GPG key
  ansible.builtin.get_url:
    url: https://kanidm.github.io/kanidm_ppa/kanidm_ppa.asc
    dest: /etc/apt/trusted.gpg.d/kanidm_ppa.asc
    mode: '0644'
#+end_src
***** Add Kanidm APT repository
#+begin_src yaml
- name: system | kanidm | add Kanidm APT repository (deb822 format)
  ansible.builtin.deb822_repository:
    name: kanidm
    types: deb
    architectures: amd64
    uris: https://kanidm.github.io/kanidm_ppa
    suites: "{{ ansible_facts['distribution_release'] | lower }}"
    components: stable
    signed_by: /etc/apt/trusted.gpg.d/kanidm_ppa.asc
    state: present
#+end_src
**** Install Client
#+begin_src yaml
- name: system | kanidm | install kanidm-unixd
  ansible.builtin.apt:
    name: kanidm-unixd
    state: present
    update_cache: yes
#+end_src
**** Configure global client
#+begin_src yaml
- name: system | kanidm | configure Kanidm global client config
  ansible.builtin.template:
    src: config.j2
    dest: /etc/kanidm/config
    mode: '0644'
#+end_src
**** Fetch unixd Account Token
#+begin_src yaml
- name: system | kanidm | fetch unixd token directly from Kanidm server
  ansible.builtin.slurp:
    src: /root/kanidm_unixd_token.txt
  delegate_to: kanidm
  register: remote_unixd_token
  no_log: true
#+end_src
**** Copy unixd Account Token
#+begin_src yaml
- name: system | kanidm | install unixd service account token
  ansible.builtin.copy:
    content: "{{ remote_unixd_token['content'] | b64decode }}"
    dest: /etc/kanidm/unixd_token
    owner: root
    group: root
    mode: '0640'
  no_log: true
  notify: system | kanidm | restart kanidm-unixd
#+end_src
**** Configure unixd daemon
#+begin_src yaml
- name: system | kanidm | configure Kanidm unix daemon config
  ansible.builtin.template:
    src: unixd.j2
    dest: /etc/kanidm/unixd
    mode: '0644'
  notify: system | kanidm | restart kanidm-unixd
#+end_src
***** Create systemd override directory
#+begin_src yaml
- name: system | kanidm | create systemd override directory for kanidm-unixd
  ansible.builtin.file:
    path: /etc/systemd/system/kanidm-unixd.service.d
    state: directory
    mode: '0755'
#+end_src

**** Create user
#+begin_src yaml
- name: users | ansible | create user
  tags: ansible,users
  user:
    name: ansible
    group: ansible
    groups: "{{ sudo_group }}"
    append: yes
    createhome: yes
    state: present
    comment: "Ansible"
    system: yes
    shell: /bin/bash
#+end_src
**** Add sudoers file
#+begin_src yaml
- name: users | ansible | add sudoers file
  tags: ansible,users,sudo
  copy:
    src: files/users/sudoers_ansible
    dest: /etc/sudoers.d/ansible
    owner: root
    group: root
    mode: 0440
    validate: "/usr/bin/visudo -cf %s"

#+end_src
*** Restart kanidm-unixd service
#+begin_src yaml
- name: system | kanidm | restart kanidm-unixd
  ansible.builtin.systemd:
    name: "{{ item }}"
    state: restarted
    enabled: true
    daemon_reload: true
  loop:
    - kanidm-unixd
    - kanidm-unixd-tasks
#+end_src
*** Restart sshd service
#+begin_src yaml
- name: system | kanidm | restart sshd
  ansible.builtin.systemd:
    name: sshd
    state: restarted
#+end_src

#+begin_src yaml
- name: system | networking | reload NetworkManager connection
  ansible.builtin.command: nmcli connection up "{{ nm_active_conn.stdout }}"
