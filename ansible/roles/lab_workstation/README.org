**** Deny incoming, Allow outgoing
#+begin_src yaml
- name: system | firewall | set default deny incoming, allow outgoing
  community.general.ufw:
    default: "{{ item.policy }}"
    direction: "{{ item.dir }}"
  loop:
    - { dir: incoming, policy: deny }
    - { dir: outgoing, policy: allow }

*** Networking configuration
:PROPERTIES:
:header-args+: :tangle tasks/system/networking.yml
:END:
#+begin_src yaml
---
#+end_src
**** Get Active Connection
#+begin_src yaml :padline no
- name: system | networking | get active NetworkManager connection
  ansible.builtin.shell: nmcli -t -f NAME connection show --active | head -n 1
  register: nm_active_conn
  changed_when: false
#+end_src
**** Configure Persistent DNS and Routing
#+begin_src yaml
- name: system | networking | configure persistent DNS and static route
  community.general.nmcli:
    conn_name: "{{ nm_active_conn.stdout }}"
    type: ethernet
    dns4:
      - "{{ ip_technitium_dns1 }}"
      - "{{ ip_technitium_dns2 }}"
    dns4_ignore_auto: true
    routes4:
      - "10.10.20.0/24 143.106.45.162"
    state: present
  when: nm_active_conn.stdout != ""
  notify: system | networking | reload NetworkManager connection
#+end_src
**** Apply Route Immediately
#+begin_src yaml
- name: system | networking | apply static route immediately to current session
  ansible.builtin.command: ip route add 10.10.20.0/24 via 143.106.45.162
  register: ip_route
  changed_when: ip_route.rc == 0
  failed_when:
    - ip_route.rc != 0
    - "'File exists' not in ip_route.stderr"
#+end_src
**** Trust Caddy Internal Root CA
***** Fetch Certificate from Caddy
#+begin_src yaml
- name: system | ca | fetch Caddy internal root certificate
  ansible.builtin.slurp:
    src: /var/lib/caddy/.local/share/caddy/pki/authorities/local/root.crt
  delegate_to: caddy
  register: caddy_root_ca_raw
  run_once: true
#+end_src
***** Install Certificate on Workstation
#+begin_src yaml
- name: system | ca | install Caddy root certificate
  ansible.builtin.copy:
    content: "{{ caddy_root_ca_raw.content | b64decode }}"
    dest: "{{ ca_cert_dir }}/caddy-internal-root.crt"
    owner: root
    group: root
    mode: '0644'
  notify: system | ca | update ca certificates
#+end_src
***** Create Brave Policies Directory
#+begin_src yaml
- name: system | brave | ensure managed policies directory exists
  ansible.builtin.file:
    path: /etc/brave/policies/managed
    state: directory
    owner: root
    group: root
    mode: '0755'
#+end_src
***** Import Certificate on Brave Browser
#+begin_src yaml
- name: system | brave | force trust of system root CAs via policy
  ansible.builtin.copy:
    dest: /etc/brave/policies/managed/ca_trust.json
    owner: root
    group: root
    mode: '0644'
    content: |
      {
        "ImportEnterpriseRoots": true
      }

**** Install APT Transport
#+begin_src yaml :padline no
- name: system | kanidm | install APT transport and dependencies
  ansible.builtin.apt:
    name: apt-transport-https
    state: present
    update_cache: yes
#+end_src
**** Configure APT Repository
***** Download Kanidm GPG key
#+begin_src yaml
- name: system | kanidm | download Kanidm GPG key
  ansible.builtin.get_url:
    url: https://kanidm.github.io/kanidm_ppa/kanidm_ppa.asc
    dest: /etc/apt/trusted.gpg.d/kanidm_ppa.asc
    mode: '0644'
#+end_src
***** Add Kanidm APT repository
#+begin_src yaml
- name: system | kanidm | add Kanidm APT repository (deb822 format)
  ansible.builtin.deb822_repository:
    name: kanidm
    types: deb
    architectures: amd64
    uris: https://kanidm.github.io/kanidm_ppa
    suites: "{{ ansible_facts['distribution_release'] | lower }}"
    components: stable
    signed_by: /etc/apt/trusted.gpg.d/kanidm_ppa.asc
    state: present
#+end_src
**** Install Client
#+begin_src yaml
- name: system | kanidm | install kanidm-unixd
  ansible.builtin.apt:
    name: kanidm-unixd
    state: present
    update_cache: yes
#+end_src
**** Configure global client
#+begin_src yaml
- name: system | kanidm | configure Kanidm global client config
  ansible.builtin.template:
    src: config.j2
    dest: /etc/kanidm/config
    mode: '0644'
#+end_src
**** Fetch unixd Account Token
#+begin_src yaml
- name: system | kanidm | fetch unixd token directly from Kanidm server
  ansible.builtin.slurp:
    src: /root/kanidm_unixd_token.txt
  delegate_to: kanidm
  register: remote_unixd_token
  no_log: true
#+end_src
**** Copy unixd Account Token
#+begin_src yaml
- name: system | kanidm | install unixd service account token
  ansible.builtin.copy:
    content: "{{ remote_unixd_token['content'] | b64decode }}"
    dest: /etc/kanidm/unixd_token
    owner: root
    group: root
    mode: '0640'
  no_log: true
  notify: system | kanidm | restart kanidm-unixd
#+end_src
**** Configure unixd daemon
#+begin_src yaml
- name: system | kanidm | configure Kanidm unix daemon config
  ansible.builtin.template:
    src: unixd.j2
    dest: /etc/kanidm/unixd
    mode: '0644'
  notify: system | kanidm | restart kanidm-unixd
#+end_src
***** Create systemd override directory
#+begin_src yaml
- name: system | kanidm | create systemd override directory for kanidm-unixd
  ansible.builtin.file:
    path: /etc/systemd/system/kanidm-unixd.service.d
    state: directory
    mode: '0755'
#+end_src
***** Configure LoadCredential
#+begin_src yaml
- name: system | kanidm | configure systemd LoadCredential for unixd token
  ansible.builtin.copy:
    content: |
      [Service]
      LoadCredential=unixd_token:/etc/kanidm/unixd_token
      Environment=KANIDM_SERVICE_ACCOUNT_TOKEN_PATH=%d/unixd_token
    dest: /etc/systemd/system/kanidm-unixd.service.d/override.conf
    mode: '0644'
  notify:
    - system | kanidm | reload systemd
    - system | kanidm | restart kanidm-unixd
#+end_src
**** Start Services
#+begin_src yaml
- name: system | kanidm | ensure daemons are started and enabled
  ansible.builtin.systemd:
    name: "{{ item }}"
    state: started
    enabled: true
  loop:
    - kanidm-unixd
    - kanidm-unixd-tasks
#+end_src
**** Enable PAM and NSS Hooks
#+begin_src yaml
- name: system | kanidm | hook Kanidm into PAM and NSS
  ansible.builtin.command: pam-auth-update --enable kanidm mkhomedir
  register: pam_update
  changed_when: false

- name: system | kanidm | add kanidm to nsswitch.conf
  ansible.builtin.replace:
    path: /etc/nsswitch.conf
    regexp: '^((passwd|group):\s+(?!.*kanidm).*)$'
    replace: '\1 kanidm'
#+end_src
**** Configure SSH Key Distribution
#+begin_src yaml
- name: system | kanidm | configure SSH to fetch Kanidm keys
  ansible.builtin.blockinfile:
    path: /etc/ssh/sshd_config
    marker: "# {mark} ANSIBLE MANAGED BLOCK - KANIDM SSH KEYS"
    block: |
      AuthorizedKeysCommand /usr/sbin/kanidm_ssh_authorizedkeys %u
      AuthorizedKeysCommandUser nobody
  notify: system | kanidm | restart sshd
#+end_src
