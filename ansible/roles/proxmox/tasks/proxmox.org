:END:

*** Configure vmbr1
#+begin_src yaml :padline no
- name: core | proxmox | configure vmbr1 NAT bridge
  ansible.builtin.blockinfile:
    path: /etc/network/interfaces
    marker: "# {mark} ANSIBLE MANAGED BLOCK: VMBR1 NAT"
    block: "{{ lookup('file', 'vmbr1.txt') }}"
  register: network_config_vmbr1
#+end_src
*** Bring up vmbr1
#+begin_src yaml
- name: core | proxmox | bring up vmbr1 interface
  ansible.builtin.command: ifup vmbr1
  when: network_config_vmbr1.changed
#+end_src
*** Configure vmbr2
#+begin_src yaml
- name: core | proxmox | configure vmbr2 VLAN-aware trunk

  register: network_config_vmbr2
#+end_src
*** Bring up vmbr2
#+begin_src yaml
- name: core | proxmox | bring up vmbr2 interface
  ansible.builtin.command: ifup vmbr2
  when: network_config_vmbr2.changed

*** Remove Enterprise Repositories
#+begin_src yaml :padline no
- name: core | proxmox | remove enterprise repositories
  ansible.builtin.file:
    path: "/etc/apt/sources.list.d/{{ item }}"
    state: absent
  loop:
    - pve-enterprise.sources
    - ceph.sources
#+end_src
*** Add PVE No-Subscription Repository
#+begin_src yaml
- name: core | proxmox | add Proxmox no-subscription repository
  ansible.builtin.deb822_repository:
    name: pve-no-subscription
    types: [deb]
    uris: "http://download.proxmox.com/debian/pve"
    suites: [trixie]
    components: [pve-no-subscription]
    state: present
#+end_src

*** Update template list
#+begin_src yaml :padline no
- name: core | proxmox | update LXC appliance template list
  ansible.builtin.command:
    cmd: pveam update
  changed_when: false
#+end_src
*** Download LXC templates
#+begin_src yaml
- name: core | proxmox | download LXC templates
  ansible.builtin.command:
    cmd: "pveam download local {{ item }}"
  args:
    creates: "/var/lib/vz/template/cache/{{ item }}"
  loop:
    - "{{ lxc_template_headscale }}"
#+end_src

*** Provision LXC containers and VMs
**** With Terraform
#+begin_src yaml :padline no :tangle no
- name: core | proxmox | provision lab infrastructure
  cloud.terraform.terraform:
    project_path: '/home/rafael/git/homelab/tofu'
    binary_path: tofu
    state: present
    force_init: yes
  delegate_to: localhost
  become: no
#+end_src
**** With Makefile
#+begin_src yaml :padline no
- name: core | proxmox | provision lab infrastructure
  ansible.builtin.shell: make auto-apply
  args:
    chdir: '/home/rafael/git/homelab/tofu'
  delegate_to: localhost
  become: no
#+end_src
