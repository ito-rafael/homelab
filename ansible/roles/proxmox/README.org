*** Configure vmbr0
#+begin_src yaml :padline no
- name: core | proxmox | configure vmbr0 port forwarding for Headscale
  ansible.builtin.blockinfile:
    path: /etc/network/interfaces
    insertafter: '^\s*bridge-fd 0'
    marker: "        # {mark} ANSIBLE MANAGED BLOCK: VMBR0 HEADSCALE DNAT"
    block: "{{ lookup('file', 'vmbr0_headscale.txt') }}"
  delegate_to: proxmox
  register: network_config_vmbr0
#+end_src

***** Init
#+begin_src yaml :padline no
- name: core | proxmox | initialize OpenTofu
  ansible.builtin.command:
    cmd: make init TOFU_ARGS="-input=false"
    chdir: /home/rafael/git/homelab/tofu
  delegate_to: localhost
  changed_when: false
#+end_src
***** Plan
#+begin_src yaml
- name: core | proxmox | create OpenTofu plan
  ansible.builtin.command:
    cmd: >
      make plan TOFU_ARGS="-input=false -no-color -out=/tmp/homelab.tfplan"
    chdir: /home/rafael/git/homelab/tofu
  delegate_to: localhost
  register: tofu_plan
  # since GNU Make obscures exact exit codes, we parse stdout to detect changes
  changed_when: "'No changes.' not in tofu_plan.stdout"
#+end_src
***** Apply
#+begin_src yaml
- name: core | proxmox | apply OpenTofu plan
  ansible.builtin.command:
    cmd: make apply TOFU_ARGS="-input=false -auto-approve /tmp/homelab.tfplan"
    chdir: /home/rafael/git/homelab/tofu
  delegate_to: localhost
  # dynamically skips the task if the previous step found no changes
  when: tofu_plan is changed
#+end_src

#+begin_src conf :tangle files/vmbr0_headscale.txt
        # forward TCP 80 (Let's Encrypt) to Headscale
        post-up   iptables -t nat -A PREROUTING -i vmbr0 -p tcp --dport 80 -j DNAT --to-destination 10.10.20.2:8080
        post-down iptables -t nat -D PREROUTING -i vmbr0 -p tcp --dport 80 -j DNAT --to-destination 10.10.20.2:8080

        # forward TCP 443 (HTTPS) to Headscale
        post-up   iptables -t nat -A PREROUTING -i vmbr0 -p tcp --dport 443 -j DNAT --to-destination 10.10.20.2:8443
        post-down iptables -t nat -D PREROUTING -i vmbr0 -p tcp --dport 443 -j DNAT --to-destination 10.10.20.2:8443

        # forward UDP 41641 (WireGuard) to Headscale
        post-up   iptables -t nat -A PREROUTING -i vmbr0 -p udp --dport 41641 -j DNAT --to-destination 10.10.20.2:41641
        post-down iptables -t nat -D PREROUTING -i vmbr0 -p udp --dport 41641 -j DNAT --to-destination 10.10.20.2:41641
#+end_src
** vmbr1
#+begin_src conf :tangle files/vmbr1.txt
auto vmbr1
iface vmbr1 inet static
        address 10.10.10.254/24
        bridge-ports none
        bridge-stp off
        bridge-fd 0
        post-up   echo 1 > /proc/sys/net/ipv4/ip_forward
        post-up   iptables -t nat -A POSTROUTING -s '10.10.10.0/16' -o vmbr0 -j MASQUERADE
        post-down iptables -t nat -D POSTROUTING -s '10.10.10.0/16' -o vmbr0 -j MASQUERADE
        post-up   ip route add 10.10.0.0/16 via 10.10.10.1
#+end_src
