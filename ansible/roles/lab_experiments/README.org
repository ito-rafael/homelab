#+begin_src yaml
- name: software | openrgb | write headless plugdev rules
  tags: software,openrgb

    dest: /etc/udev/rules.d/61-openrgb.rules
    owner: root
    group: root
    mode: '0644'
  notify: reload udev rules

**** Reserve Space (ext4)
#+begin_src yaml
# create a reserved block of 1% (~10 GB) to avoid disk filled
# we set "changed_when" to false, since tune2fs does not have a built-in idempotency check
- name: system | disk-protection | create reserved disk space block (ext4)
  tags: system,disk
  ansible.builtin.command: "tune2fs -m 1 {{ root_mount.device }}"
  when: root_mount.fstype == 'ext4'
  changed_when: false
#+end_src
**** Reserve Space (BTRFS)
***** Create Reserved Space
#+begin_src yaml
- name: system | disk-protection | create reserved disk space block (BTRFS)
  tags: system,disk
  ansible.builtin.command: fallocate -l 10G /root/emergency-space.img
  when: root_mount.fstype == 'btrfs'
  args:
    creates: /root/emergency-space.img
#+end_src
***** Protect Reserved Space
#+begin_src yaml
- name: system | disk-protection | protect the BTRFS emergency file from accidental deletion
  tags: system,disk
  ansible.builtin.file:
    path: /root/emergency-space.img
    state: file
    mode: '0000'
  when: root_mount.fstype == 'btrfs'

# if/when needed: ssh into the machine, then "sudo rm /root/emergency-space.img"

**** Make Mamba Available on Login
#+begin_src yaml
- name: system | mamba | make mamba available to users upon login
  tags: system,mamba
  ansible.builtin.copy:
    dest: /etc/profile.d/mamba.sh
    content: |
      export MAMBA_ROOT_PREFIX=/opt/mamba
      . /opt/mamba/etc/profile.d/conda.sh
      . /opt/mamba/etc/profile.d/mamba.sh

      # force Mamba to use the user's home for packages and environments
      export CONDA_ENVS_PATH="$HOME/.conda/envs"
      export CONDA_PKGS_DIRS="$HOME/.conda/pkgs:/opt/mamba/pkgs"

      # ensure the user's local package cache exists
      mkdir -p "$HOME/.conda/pkgs"

      # disable system-level updates for users
      export CONDA_AUTO_UPDATE_CONDA=false
      # prevent "short-name" interactive prompts
      export CONDA_SHORT_NAME_MODE=disabled
    owner: root
    group: root

#+begin_src conf :tangle files/openrgb/effects/dna-double-helix.glsl
// DNA Double Helix — OpenRGB Effects Plugin
// Smooth 4-column x 10-row Visual Map, with alternating front/back depth.
// Apply to the combined Visual Map, not to each RAM stick separately.
// Green connecting rungs are removed.

const float SPEED = 1.20;                 // Animation speed: 0 = stopped, larger = faster
const float BRIGHTNESS = 1.00;            // Overall brightness: 0 = off, 1 = maximum
const float BACK_STRAND_BRIGHTNESS = 0.28; // Brightness of the strand behind the front one
const float FRONT_HIGHLIGHT = 0.18;       // Subtle highlight on the front-facing strand

void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    vec2 res = max(iResolution.xy, vec2(1.0));
    vec2 uv = fragCoord.xy / res;

    // LED-center coordinates: columns 0..3 and rows 0..9.
    // Fractional coordinates produce smooth movement across neighboring LEDs.
    float col = clamp(uv.x * 4.0 - 0.5, 0.0, 3.0);
    float row = clamp(uv.y * 10.0 - 0.5, 0.0, 9.0);

    // The traveling phase controls both the crossing position and the depth.
    // As cos(phase) changes sign, the blue and pink strands swap front/back.
    const float TAU = 6.28318530718;
    float phase = row * (TAU / 10.0) + 1.57079632679 - iTime * SPEED;
    float wave = sin(phase);
    float depthA = cos(phase); // positive: cyan/blue strand A faces forward

    float strandPosA = 1.5 + 1.45 * wave;
    float strandPosB = 1.5 - 1.45 * wave;

    vec3 cyan       = vec3(0.03, 0.80, 1.00);
    vec3 magenta    = vec3(1.00, 0.04, 0.44);
    vec3 background = vec3(0.001, 0.003, 0.010);

    // Soft strand coverage: LEDs fade as each strand moves between columns.
    const float STRAND_WIDTH = 1.05;
    float onA = 1.0 - smoothstep(0.0, STRAND_WIDTH, abs(col - strandPosA));
    float onB = 1.0 - smoothstep(0.0, STRAND_WIDTH, abs(col - strandPosB));

    // Depth continuously changes each strand's brightness without changing its
    // identity: blue stays blue and pink stays pink. They trade depth as cos(phase)
    // changes sign, so the same-color strand is bright at the front crossings and
    // dimmer while it passes behind the other strand.
    float shadeA = BACK_STRAND_BRIGHTNESS
                 + (1.0 - BACK_STRAND_BRIGHTNESS) * (0.5 + 0.5 * depthA);
    float shadeB = BACK_STRAND_BRIGHTNESS
                 + (1.0 - BACK_STRAND_BRIGHTNESS) * (0.5 - 0.5 * depthA);

    // At the projected crossings, depthA determines the draw order. These
    // crossings happen where abs(depthA) is near 1, so the order changes while
    // the strands are spatially separated and does not visibly pop at a crossing.
    float aIsFront = step(0.0, depthA);
    float frontOn = mix(onB, onA, aIsFront);
    float backOn  = mix(onA, onB, aIsFront);
    vec3 frontColor = mix(magenta * shadeB, cyan * shadeA, aIsFront);
    vec3 backColor  = mix(cyan * shadeA, magenta * shadeB, aIsFront);

    // Render back first, then the front strand on top at the crossing.
    vec3 color = mix(background, backColor, clamp(backOn, 0.0, 1.0));
    float frontShine = 1.0 + FRONT_HIGHLIGHT * abs(depthA);
    color = mix(color, frontColor * frontShine, clamp(frontOn, 0.0, 1.0));

    // Gentle pulse retained from the original smooth animation.
    float pulse = 0.90 + 0.10 * (0.5 + 0.5 * sin((iTime * SPEED) * 2.0 + row * 0.35));
    color *= pulse * clamp(BRIGHTNESS, 0.0, 1.0);

    fragColor = vec4(clamp(color, 0.0, 1.0), 1.0);
}

#+begin_src conf :tangle files/openrgb/effects/dna-double-helix-3d.glsl
// DNA Double Helix — 3D axial rotation for OpenRGB Effects > Shaders
// For a 4-column x 10-row Visual Map (40 LEDs).
// Apply to the combined Visual Map, not to each RAM stick separately.
// No connecting rungs: only the cyan and magenta strands are rendered.
//
// The rotation is projected from a simple 3D helix:
//   horizontal position = sin(angle)
//   depth / front-back = cos(angle)
// As time advances, the angle changes so the two strands orbit the long axis;
// each crossing swaps which strand is in front. The front strand becomes wider,
// brighter, and gets a small cool-white highlight; the rear strand recedes.

const float ROTATION_SPEED = 1.20;          // Radians/second; higher = faster rotation
const float BRIGHTNESS = 1.00;               // Overall brightness: 0.0 to 1.0
const float TURN_LENGTH = 10.0;              // Rows per full helical turn
const float STRAND_RADIUS = 1.45;             // Horizontal travel in LED columns
const float BACK_BRIGHTNESS = 0.22;           // Rear strand brightness at maximum depth
const float FRONT_HIGHLIGHT = 0.18;           // Front-facing specular highlight
const float MIN_STRAND_WIDTH = 0.90;          // Rear strand width in LED units
const float MAX_STRAND_WIDTH = 1.28;          // Front strand width in LED units

const float TAU = 6.28318530718;
const float PI  = 3.14159265359;

float strandCoverage(float col, float strandX, float width)
{
    float d = abs(col - strandX);
    return 1.0 - smoothstep(0.12, width, d);
}

void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    vec2 res = max(iResolution.xy, vec2(1.0));
    vec2 uv = fragCoord.xy / res;

    // Continuous coordinates centered on the 4 x 10 LED grid.
    float col = clamp(uv.x * 4.0 - 0.5, 0.0, 3.0);
    float row = clamp(uv.y * 10.0 - 0.5, 0.0, 9.0);

    // Rotate the helix around its long axis. The sin() component is its visible
    // left/right orbit; cos() is depth, determining which strand faces camera.
    float angleA = row * (TAU / TURN_LENGTH) - iTime * ROTATION_SPEED;
    float angleB = angleA + PI;

    float depthA = cos(angleA);  // +1 = cyan strand fully forward
    float depthB = cos(angleB);  // opposite depth to strand A

    float mid = 1.5;
    float xA = mid + STRAND_RADIUS * sin(angleA);
    float xB = mid + STRAND_RADIUS * sin(angleB);

    // A strand is visually fuller when it faces the viewer and thinner as it
    // rotates away. This helps the motion read as a rotating 3D object.
    float widthA = mix(MIN_STRAND_WIDTH, MAX_STRAND_WIDTH, 0.5 + 0.5 * depthA);
    float widthB = mix(MIN_STRAND_WIDTH, MAX_STRAND_WIDTH, 0.5 + 0.5 * depthB);
    float onA = strandCoverage(col, xA, widthA);
    float onB = strandCoverage(col, xB, widthB);

    vec3 cyan = vec3(0.03, 0.80, 1.00);
    vec3 pink = vec3(1.00, 0.035, 0.42);
    vec3 background = vec3(0.001, 0.003, 0.010);

    // Depth shading: the rear side is deliberately darker, while the forward
    // side gets a subtle highlight that moves around the helix with the rotation.
    float shadeA = mix(BACK_BRIGHTNESS, 1.0, smoothstep(-0.35, 0.55, depthA));
    float shadeB = mix(BACK_BRIGHTNESS, 1.0, smoothstep(-0.35, 0.55, depthB));
    float shineA = FRONT_HIGHLIGHT * pow(max(depthA, 0.0), 8.0);
    float shineB = FRONT_HIGHLIGHT * pow(max(depthB, 0.0), 8.0);
    vec3 colorA = mix(cyan * shadeA, vec3(1.0), shineA);
    vec3 colorB = mix(pink * shadeB, vec3(1.0), shineB);

    // The strands have opposite depth. Paint the rear one first, then place the
    // front one over it. The swap occurs when their projected positions are far
    // apart, avoiding a visible popping effect at the crossing.
    float aIsFront = step(0.0, depthA);
    float backOn = mix(onA, onB, aIsFront);
    float frontOn = mix(onB, onA, aIsFront);
    vec3 backColor = mix(colorA, colorB, aIsFront);
    vec3 frontColor = mix(colorB, colorA, aIsFront);

    vec3 color = mix(background, backColor, clamp(backOn, 0.0, 1.0));
    color = mix(color, frontColor, clamp(frontOn, 0.0, 1.0));

    // Very subtle breathing highlights; the actual axial rotation remains the
    // dominant motion and both strands keep their original cyan/pink identity.
    float pulse = 0.96 + 0.04 * (0.5 + 0.5 * sin(iTime * ROTATION_SPEED * 1.7 + row * 0.28));
    color *= pulse * clamp(BRIGHTNESS, 0.0, 1.0);

    fragColor = vec4(clamp(color, 0.0, 1.0), 1.0);
}
