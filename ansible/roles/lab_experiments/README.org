#+end_src
**** Reserve Space (ext4)
#+begin_src yaml
# create a reserved block of 1% (~10 GB) to avoid disk filled
# we set "changed_when" to false, since tune2fs does not have a built-in idempotency check
- name: system | disk-protection | create reserved disk space block (ext4)
  tags: system,disk
  ansible.builtin.command: "tune2fs -m 1 {{ root_mount.device }}"
  when: root_mount.fstype == 'ext4'
  changed_when: false
#+end_src
**** Reserve Space (BTRFS)
***** Create Reserved Space
#+begin_src yaml
- name: system | disk-protection | create reserved disk space block (BTRFS)
  tags: system,disk
  ansible.builtin.command: fallocate -l 10G /root/emergency-space.img
  when: root_mount.fstype == 'btrfs'
  args:
    creates: /root/emergency-space.img
#+end_src
***** Protect Reserved Space
#+begin_src yaml
- name: system | disk-protection | protect the BTRFS emergency file from accidental deletion
  tags: system,disk
  ansible.builtin.file:
    path: /root/emergency-space.img
    state: file
    mode: '0000'
  when: root_mount.fstype == 'btrfs'

# if/when needed: ssh into the machine, then "sudo rm /root/emergency-space.img"

**** Make Mamba Available on Login
#+begin_src yaml
- name: system | mamba | make mamba available to users upon login
  tags: system,mamba
  ansible.builtin.copy:
    dest: /etc/profile.d/mamba.sh
    content: |
      export MAMBA_ROOT_PREFIX=/opt/mamba
      . /opt/mamba/etc/profile.d/conda.sh
      . /opt/mamba/etc/profile.d/mamba.sh

      # force Mamba to use the user's home for packages and environments
      export CONDA_ENVS_PATH="$HOME/.conda/envs"
      export CONDA_PKGS_DIRS="$HOME/.conda/pkgs:/opt/mamba/pkgs"

      # ensure the user's local package cache exists
      mkdir -p "$HOME/.conda/pkgs"

      # disable system-level updates for users
      export CONDA_AUTO_UPDATE_CONDA=false
      # prevent "short-name" interactive prompts
      export CONDA_SHORT_NAME_MODE=disabled
    owner: root
    group: root
    mode: '0644'
#+end_src
**** Configure Global condarc
#+begin_src yaml
- name: system | mamba | config global condarc for user envs
  tags: system,mamba
  ansible.builtin.copy:
    dest: /opt/mamba/.condarc
    content: |
      # set default channels
      channels:
        - conda-forge

      # route new envs and cached packages to the user's home dir
      envs_dirs:
        - ~/.conda/envs
        - /opt/mamba/envs
      pkgs_dirs:
        - ~/.conda/pkgs
        - /opt/mamba/pkgs

      short-name-mode: disabled
    owner: root
    group: root
    mode: '0644'
#+end_src

**** Check Architecture
#+begin_src yaml :padline no
# NVIDIA GPU architecture identification
#+end_src
***** Kepler
#+begin_src yaml
- name: system | nvidia | check if architecture is Kepler
  tags: system,gpu,nvidia,nvidia-gpu
  shell: lspci -k -d ::03xx | grep -E 'GK[12][01][0-9A][B]?'
  changed_when: false
  failed_when: false
  register: kepler
#+end_src
***** Maxwell, Pascall, Volta
#+begin_src yaml
- name: system | nvidia | check if architecture is Maxwell, Pascal, or Volta
  tags: system,gpu,nvidia,nvidia-gpu
  shell: lspci -k -d ::03xx | grep -E 'GM[12]0[04678B]|GP10[24678]|GV100'
  changed_when: false
  failed_when: false
  register: maxwell_pascal_volta
#+end_src
***** Turing, Ampere, Ada Lovelace
#+begin_src yaml
- name: system | nvidia | check if architecture is Turing, Ampere, or Ada Lovelace
  tags: system,gpu,nvidia,nvidia-gpu
  shell: lspci -k -d ::03xx | grep -E 'TU1[01][2467]|GA10[2467]AD10[23467]'
  changed_when: false
  failed_when: false
  register: turing_ampere_adalovelace
#+end_src
***** Blackwell
#+begin_src yaml
- name: system | nvidia | check if architecture is Blackwell
  tags: system,gpu,nvidia,nvidia-gpu
  shell: lspci -k -d ::03xx | grep -E 'GB[12][0-9]{2}'
  changed_when: false
  failed_when: false
  register: blackwell
#+end_src

:header-args+: :tangle tasks/system/podman.yml
:END:
#+begin_src yaml
---
#+end_src
**** Uninstall Docker
#+begin_src yaml :padline no
- name: system | podman | prevent packages conflict
  tags: system,podman
  package:
    name: docker
    state: absent
#+end_src
**** Install Podman
#+begin_src yaml
- name: system | podman | install Podman
  tags: system,podman
  package:
    name:
      - podman          # tool and library to run OCI-based containers in pods
      - podman-compose  # script to run docker-compose.yml
      - podman-docker   # aliases to emulate Docker CLI

#+end_src
**** Enable User Podman Socket
#+begin_src yaml :tangle no
- name: system | podman | enable user podman socket
  tags: system,podman
  ansible.builtin.systemd_service:
    name: podman.socket
    scope: user
    enabled: true
    state: started
#+end_src
**** Export DOCKER_HOST
#+begin_src yaml :tangle no
- name: system | podman | export DOCKER_HOST in shell profile
  tags: system,podman
  ansible.builtin.lineinfile:
    path: /etc/profile.d/podman-docker-host.sh
    line: 'export DOCKER_HOST="unix://${XDG_RUNTIME_DIR}/podman/podman.sock"'
    create: true
    mode: '0644'
#+end_src
**** Test GPU Support
#+begin_src yaml
- name: system | podman | [debug] test GPU support Podman containers
  tags: system,podman
  ansible.builtin.command: podman run --rm --device nvidia.com/gpu=all ubuntu nvidia-smi
  #ansible.builtin.command: podman run --rm --device nvidia.com/gpu=all ubuntu nvidia-smi -L
  register: podman_gpu_test
  changed_when: false
#+end_src
**** Display nvidia-smi Command Output
#+begin_src yaml
- name: system | podman | [debug] display the output of nvidia-smi inside container
  tags: system,podman
  ansible.builtin.debug:
    var: podman_gpu_test.stdout_lines
#+end_src
**** Configure Registries
#+begin_src yaml
- name: system | podman | configure unqualified-search-registries
  tags: system,podman
  ansible.builtin.lineinfile:
    path: /etc/containers/registries.conf
    regexp: 'unqualified-search-registries ='
    line: 'unqualified-search-registries = ["docker.io", "quay.io"]'
    state: present
#+end_src
**** Configure short-name-mode
#+begin_src yaml
- name: system | podman | configure short-name-mode
  tags: system,podman
  ansible.builtin.lineinfile:
    path: /etc/containers/registries.conf
    regexp: 'short-name-mode ='
    line: 'short-name-mode = "disabled"'
    state: present
#+end_src
