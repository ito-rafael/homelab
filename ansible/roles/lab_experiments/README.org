**** Reserve Space (ext4)
#+begin_src yaml
# create a reserved block of 1% (~10 GB) to avoid disk filled
# we set "changed_when" to false, since tune2fs does not have a built-in idempotency check
- name: system | disk-protection | create reserved disk space block (ext4)
  tags: system,disk
  ansible.builtin.command: "tune2fs -m 1 {{ root_mount.device }}"
  when: root_mount.fstype == 'ext4'
  changed_when: false
#+end_src
**** Reserve Space (BTRFS)
***** Create Reserved Space
#+begin_src yaml
- name: system | disk-protection | create reserved disk space block (BTRFS)
  tags: system,disk
  ansible.builtin.command: fallocate -l 10G /root/emergency-space.img
  when: root_mount.fstype == 'btrfs'
  args:
    creates: /root/emergency-space.img
#+end_src
***** Protect Reserved Space
#+begin_src yaml
- name: system | disk-protection | protect the BTRFS emergency file from accidental deletion
  tags: system,disk
  ansible.builtin.file:
    path: /root/emergency-space.img
    state: file
    mode: '0000'
  when: root_mount.fstype == 'btrfs'

# if/when needed: ssh into the machine, then "sudo rm /root/emergency-space.img"

**** Make Mamba Available on Login
#+begin_src yaml
- name: system | mamba | make mamba available to users upon login
  tags: system,mamba
  ansible.builtin.copy:
    dest: /etc/profile.d/mamba.sh
    content: |
      export MAMBA_ROOT_PREFIX=/opt/mamba
      . /opt/mamba/etc/profile.d/conda.sh
      . /opt/mamba/etc/profile.d/mamba.sh

      # force Mamba to use the user's home for packages and environments
      export CONDA_ENVS_PATH="$HOME/.conda/envs"
      export CONDA_PKGS_DIRS="$HOME/.conda/pkgs:/opt/mamba/pkgs"

      # ensure the user's local package cache exists
      mkdir -p "$HOME/.conda/pkgs"

      # disable system-level updates for users
      export CONDA_AUTO_UPDATE_CONDA=false
      # prevent "short-name" interactive prompts
      export CONDA_SHORT_NAME_MODE=disabled
    owner: root
    group: root
    mode: '0644'
#+end_src

**** Check Architecture
#+begin_src yaml :padline no
# NVIDIA GPU architecture identification
#+end_src
***** Kepler
#+begin_src yaml
- name: system | nvidia | check if architecture is Kepler
  tags: system,gpu,nvidia,nvidia-gpu
  shell: lspci -k -d ::03xx | grep -E 'GK[12][01][0-9A][B]?'
  changed_when: false
  failed_when: false
  register: kepler
#+end_src
***** Maxwell, Pascall, Volta
#+begin_src yaml
- name: system | nvidia | check if architecture is Maxwell, Pascal, or Volta
  tags: system,gpu,nvidia,nvidia-gpu
  shell: lspci -k -d ::03xx | grep -E 'GM[12]0[04678B]|GP10[24678]|GV100'
  changed_when: false
  failed_when: false
  register: maxwell_pascal_volta
#+end_src
***** Turing, Ampere, Ada Lovelace
#+begin_src yaml
- name: system | nvidia | check if architecture is Turing, Ampere, or Ada Lovelace
  tags: system,gpu,nvidia,nvidia-gpu
  shell: lspci -k -d ::03xx | grep -E 'TU1[01][2467]|GA10[2467]AD10[23467]'
  changed_when: false
  failed_when: false
  register: turing_ampere_adalovelace
#+end_src
***** Blackwell
#+begin_src yaml
- name: system | nvidia | check if architecture is Blackwell
  tags: system,gpu,nvidia,nvidia-gpu
  shell: lspci -k -d ::03xx | grep -E 'GB[12][0-9]{2}'
  changed_when: false
  failed_when: false
  register: blackwell
#+end_src

#+begin_src conf :tangle files/openrgb/dna_double_helix.glsl
// DNA Double Helix — OpenRGB Effects Plugin
// Tuned for a Visual Map with 10 rows x 4 columns (40 LEDs).
// Apply to the combined Visual Map, not to each RAM stick separately.
//
// User-adjustable parameters:
const float SPEED = 1.20;       // Animation speed: 0.0 = stopped, larger = faster
const float BRIGHTNESS = 1.00;  // Overall LED brightness: 0.0 = off, 1.0 = maximum
//
// Smooth-transition version:
// Strand positions remain fractional instead of snapping to whole columns.
// Each LED's intensity is blended by its distance from each strand so adjacent
// LEDs fade down/up continuously as the strands move across the 4-column matrix.

void mainImage(out vec4 fragColor, in vec2 fragCoord)
{
    vec2 res = max(iResolution.xy, vec2(1.0));
    vec2 uv = fragCoord.xy / res;

    // LED-center coordinates: columns 0..3 and rows 0..9.
    // Keeping fractional coordinates is important for smooth cross-fading.
    float col = clamp(uv.x * 4.0 - 0.5, 0.0, 3.0);
    float row = clamp(uv.y * 10.0 - 0.5, 0.0, 9.0);

    // One full turn along the 10 LEDs of a stick.
    float phase = row * (6.28318530718 / 10.0)
                + 1.57079632679
                - iTime * SPEED;
    float wave = sin(phase);

    // Fractional strand positions. Do not round these to integer columns.
    float strandPosA = 1.5 + 1.45 * wave;
    float strandPosB = 1.5 - 1.45 * wave;

    vec3 cyan       = vec3(0.03, 0.80, 1.00);
    vec3 magenta    = vec3(1.00, 0.04, 0.44);
    vec3 rungGreen  = vec3(0.18, 1.00, 0.48);
    vec3 background = vec3(0.001, 0.003, 0.010);

    // Smoothly fade each strand across nearby LEDs instead of switching LEDs
    // abruptly. A strand at an LED center has full strength; as it moves toward
    // the next column, one LED dims while the next one brightens.
    const float STRAND_WIDTH = 1.05;
    float onA = 1.0 - smoothstep(0.0, STRAND_WIDTH, abs(col - strandPosA));
    float onB = 1.0 - smoothstep(0.0, STRAND_WIDTH, abs(col - strandPosB));

    // Soft connecting rungs between the two fractional strand positions.
    // Their ends also fade rather than appearing/disappearing as hard steps.
    float lo = min(strandPosA, strandPosB);
    float hi = max(strandPosA, strandPosB);
    float rungStart = smoothstep(lo + 0.10, lo + 0.75, col);
    float rungEnd   = 1.0 - smoothstep(hi - 0.75, hi - 0.10, col);
    float rung = clamp(rungStart * rungEnd, 0.0, 1.0);

    // Blend strand colors smoothly near crossings. This weighted blend also
    // avoids one strand abruptly hiding the other at a crossover.
    float strandTotal = onA + onB;
    vec3 strandColor = (cyan * onA + magenta * onB) / max(strandTotal, 0.0001);
    float strandStrength = clamp(strandTotal, 0.0, 1.0);

    vec3 color = mix(background, rungGreen, rung * 0.75);
    color = mix(color, strandColor, strandStrength);

    // Gentle pulse; its timing follows SPEED as well.
    float pulse = 0.90 + 0.10 * (0.5 + 0.5 * sin((iTime * SPEED) * 2.0 + row * 0.35));
    color *= pulse;

    // Apply overall brightness after all color/pulse calculations.
    color *= clamp(BRIGHTNESS, 0.0, 1.0);

    fragColor = vec4(clamp(color, 0.0, 1.0), 1.0);
}
